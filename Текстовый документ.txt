import arcade
import random
import time

WINDOW_WIDTH = 1280
WINDOW_HEIGHT = 720
WINDOW_TITLE = 'Башня теней'

TILE_SCALING = 0.5
COIN_SCALING = 0.5
PLAYER_SCALING = 0.2

PLAYER_MOVEMENT_SPEED = 10
GRAVITY = 1
PLAYER_JUMP_SPEED = 20

ATTACK_RANGE = 200




class Enemy(arcade.Sprite):
    def __init__(self, walls):
        super().__init__("enemy.png", scale=0.3)
        self.center_x = random.randint(100, 1000)
        self.center_y = random.randint(300, 500)
        self.change_x = random.choice([-2, 2])
        self.walls = walls
        self.physics_engine = arcade.PhysicsEnginePlatformer(self, walls=self.walls, gravity_constant=GRAVITY)
        self.jump_chance = 0.01

    def update(self, delta_time: float = 1 / 60):
        self.physics_engine.update()
        self.center_x += self.change_x

        front_x = self.center_x + (self.width // 2 + 2) * (1 if self.change_x > 0 else -1)
        front_y = self.center_y - self.height // 2 - 5
        check_sprite = arcade.Sprite(center_x=front_x, center_y=front_y)
        if not arcade.check_for_collision_with_list(check_sprite, self.walls):
            self.change_x *= -1

        if random.random() < self.jump_chance:
            if self.physics_engine.can_jump():
                self.change_y = PLAYER_JUMP_SPEED


class GameView(arcade.View):
    def __init__(self):
        super().__init__()

        self.player_textures = [
            arcade.load_texture("run1.png"),
            arcade.load_texture("run2.png"),
            arcade.load_texture("run3.png"),
        ]
        self.current_texture = 0
        self.animation_speed = 0.1
        self.animation_timer = 0

        self.collect_coin_sound = arcade.load_sound(":resources:sounds/coin1.wav")
        self.jump_sound = arcade.load_sound(":resources:sounds/jump1.wav")

        self.score = 0
        self.enemies_killed = 0
        self.player_health = 3

        self.score_text = arcade.Text(f"Score: {self.score}", x=0, y=5)
        self.health_text = arcade.Text(f"Health: {self.player_health}", x=0, y=45)
        self.kills_text = arcade.Text(f"Kills: {self.enemies_killed}", x=0, y=25)

        self.enemy_spawn_timer = 0
        self.enemy_spawn_interval = 5

        self.is_attacking = False
        self.attack_cooldown = 0.5
        self.last_attack_time = 0

        self.goal_x = 2500
        self.goal_y = 80

    def setup(self):

        arcade.set_background_color(arcade.color.SKY_BLUE)

        layer_options = {
            "wall2": {"use_spatial_hash": True},
            "platform": {"use_spatial_hash": True},
            "platform1": {"use_spatial_hash": True}
        }
        self.tile_map = arcade.load_tilemap("game_map.json", scaling=TILE_SCALING, layer_options=layer_options)
        self.scene = arcade.Scene.from_tilemap(self.tile_map)

        self.player_sprite = arcade.Sprite(self.player_textures[0], scale=PLAYER_SCALING)
        self.player_sprite.center_x = 64
        self.player_sprite.center_y = 200
        self.scene.add_sprite("Player", self.player_sprite)

        platforms = arcade.SpriteList()
        platforms.extend(self.scene["wall"])
        platforms.extend(self.scene["platform"])
        platforms.extend(self.scene["platform1"])

        self.physics_engine = arcade.PhysicsEnginePlatformer(
            self.player_sprite, walls=platforms, gravity_constant=GRAVITY
        )

        self.coins = arcade.SpriteList()
        positions = [(300, 200), (500, 300), (700, 400), (900, 200), (1100, 250)]
        for pos in positions:
            coin = arcade.Sprite("coin.png", scale=COIN_SCALING)
            coin.center_x = pos[0]
            coin.center_y = pos[1]
            self.coins.append(coin)

        self.scene.add_sprite_list("Coins", sprite_list=self.coins)

        self.enemies = arcade.SpriteList()
        for _ in range(3):
            self.spawn_enemy()
        self.scene.add_sprite_list("Enemies", sprite_list=self.enemies)

        goal = arcade.Sprite("flag.png", scale=0.1)
        goal.center_x = self.goal_x
        goal.center_y = self.goal_y
        self.scene.add_sprite("Goal", goal)

        self.camera = arcade.camera.Camera2D()
        self.gui_camera = arcade.camera.Camera2D()

    def spawn_enemy(self):
        enemy = Enemy(walls=self.scene["wall"])
        self.enemies.append(enemy)

    def attack(self):
        for enemy in self.enemies:
            distance = arcade.get_distance_between_sprites(self.player_sprite, enemy)
            if distance <= ATTACK_RANGE:
                enemy.remove_from_sprite_lists()
                self.enemies_killed += 1
                self.kills_text.text = f"Kills: {self.enemies_killed}"

    def on_draw(self):
        self.clear()
        self.camera.use()
        self.scene.draw()
        self.gui_camera.use()
        self.score_text.draw()
        self.health_text.draw()
        self.kills_text.draw()

    def on_update(self, delta_time):
        self.physics_engine.update()

        if self.player_sprite.change_x != 0:
            self.animation_timer += delta_time
            if self.animation_timer >= self.animation_speed:
                self.animation_timer = 0
                self.current_texture = (self.current_texture + 1) % len(self.player_textures)
            self.player_sprite.texture = self.player_textures[self.current_texture]
        else:
            self.player_sprite.texture = self.player_textures[0]

        coin_hit_list = arcade.check_for_collision_with_list(self.player_sprite, self.scene["Coins"])
        for coin in coin_hit_list:
            coin.remove_from_sprite_lists()
            arcade.play_sound(self.collect_coin_sound)
            self.score += 1
            self.score_text.text = f"Score: {self.score}"

        self.enemies.update()

        for enemy in arcade.check_for_collision_with_list(self.player_sprite, self.enemies):
            self.player_health -= 1
            self.health_text.text = f"Health: {self.player_health}"
            enemy.remove_from_sprite_lists()
            if self.player_health <= 0:
                game_over_view = GameOverView(self.score, self.enemies_killed)
                self.window.show_view(game_over_view)

        self.enemy_spawn_timer += delta_time
        if self.enemy_spawn_timer >= self.enemy_spawn_interval:
            self.spawn_enemy()
            self.enemy_spawn_timer = 0

        if self.player_sprite.center_x >= self.goal_x and abs(self.player_sprite.center_y - self.goal_y) < 80:
            victory_view = VictoryView(self.score, self.enemies_killed)
            self.window.show_view(victory_view)

        self.camera.position = self.player_sprite.position

    def on_key_press(self, key, modifiers):
        if key == arcade.key.ESCAPE:
            self.setup()
        if key in (arcade.key.UP, arcade.key.W):
            if self.physics_engine.can_jump():
                self.player_sprite.change_y = PLAYER_JUMP_SPEED
                arcade.play_sound(self.jump_sound)
        elif key in (arcade.key.LEFT, arcade.key.A):
            self.player_sprite.change_x = -PLAYER_MOVEMENT_SPEED
        elif key in (arcade.key.RIGHT, arcade.key.D):
            self.player_sprite.change_x = PLAYER_MOVEMENT_SPEED
        elif key == arcade.key.SPACE:
            current_time = time.time()
            if current_time - self.last_attack_time > self.attack_cooldown:
                self.is_attacking = True
                self.last_attack_time = current_time
                self.attack()

    def on_key_release(self, key, modifiers):
        if key in (arcade.key.UP, arcade.key.W, arcade.key.DOWN, arcade.key.S):
            self.player_sprite.change_y = 0
        elif key in (arcade.key.LEFT, arcade.key.A, arcade.key.RIGHT, arcade.key.D):
            self.player_sprite.change_x = 0


class GameOverView(arcade.View):
    def __init__(self, score, enemies_killed):
        super().__init__()
        self.score = score
        self.enemies_killed = enemies_killed

    def on_show_view(self):
        arcade.set_background_color(arcade.color.BLACK)

    def on_draw(self):
        self.clear()
        arcade.draw_text("Game Over", self.window.width / 2, self.window.height / 2 + 100,
                         arcade.color.WHITE, font_size=40, anchor_x="center")
        arcade.draw_text(f"Coins Collected: {self.score}", self.window.width / 2, self.window.height / 2 + 40,
                         arcade.color.YELLOW, font_size=24, anchor_x="center")
        arcade.draw_text(f"Enemies Defeated: {self.enemies_killed}", self.window.width / 2, self.window.height / 2,
                         arcade.color.RED, font_size=24, anchor_x="center")
        arcade.draw_text("Press ENTER to Restart", self.window.width / 2, self.window.height / 2 - 100,
                         arcade.color.GRAY, font_size=20, anchor_x="center")

    def on_key_press(self, key, modifiers):
        if key == arcade.key.ENTER:
            game_view = GameView()
            game_view.window = self.window
            game_view.setup()
            self.window.show_view(game_view)
